// Cache for statistics data
private StatisticsData cachedStatisticsData = null;

// ============== ADD AFTER CONSTRUCTOR ==============

/**
 * Called by StatisticsLoadWorker.done() when data is loaded
 * Updates UI from cached data on EDT (safe from multiple threads)
 */
public void updateUIWithData(StatisticsData data) {
    // Cache the data
    this.cachedStatisticsData = data;
    
    // Critical: Update UI on EDT
    javax.swing.SwingUtilities.invokeLater(() -> {
        System.out.println("📊 Updating statistics panel UI from cached data");
        
        try {
            // Now call the original methods - they will use cached data
            initUI();
            addControl();
            
            System.out.println("✅ Statistics panel updated successfully");
        } catch (Exception e) {
            System.out.println("❌ Error updating statistics: " + e.getMessage());
            e.printStackTrace();
        }
    });
}

/**
 * Helper: Get cached statistics data if available, otherwise query DB
 * This allows existing code to work with both cached and live data
 */
public int getCachedTongTienHoaDonNgay(java.sql.Date date) {
    if (cachedStatisticsData != null) {
        return cachedStatisticsData.dthungay;
    }
    return TK.getTongTienHoaDonngay(date);
}

public int getCachedSoLuongHoaDon(java.sql.Date date) {
    if (cachedStatisticsData != null) {
        return Integer.parseInt(cachedStatisticsData.soHoaDon + "");
    }
    return TK.getsoluongHD(date);
}
